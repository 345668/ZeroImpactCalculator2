import datetime

# Conversion factors: Fuel unit to kWh
CONVERSION_FACTORS = {
    "natural gas": 10.4,  # 1 m³ = 10.4 kWh
    "heating oil": 9.4,   # 1 L = 9.4 kWh
    "wood pellets": 4.8   # 1 kg = 4.8 kWh
}

# Emission factors (kg CO₂/kWh) for 2024
EMISSION_FACTORS = {
    "heating oil": 0.266,
    "natural gas": 0.202,
    "liquefied petroleum gas": 0.234,
    "district heating": 0.195,
    "electricity mix": 0.343,
    "coal heating": 0.351,
    "wood pellets": 0.020,
    "firewood": 0.015,
    "biogas": 0.045,
    "heat pump (electricity mix)": 0.086,  # COP 4.0
    "heat pump (green electricity)": 0.000,
    "green electricity": 0.000,
    "solar thermal": 0.000,
    "pv self-consumption": 0.000
}

CARBON_PRICE_PER_TON = 50  # EUR per ton
CREDITING_PERIOD_YEARS = 10  # Updated crediting period

def calculate_carbon_credits(
        document_id, building_type, location, area_sqm, construction_year,
        heating_system, current_fuel, current_fuel_consumption,
        future_system, future_energy_consumption, mixed_future_sources=None):
    """
    Calculate carbon credits and financial value for a building renovation project.
    """

    # Extract today's date
    date_assessed = datetime.date.today().strftime("%Y-%m-%d")

    # Convert current fuel consumption to kWh
    if current_fuel not in CONVERSION_FACTORS:
        raise ValueError(f"Invalid fuel type: {current_fuel}. Use one of: {list(CONVERSION_FACTORS.keys())}.")

    current_energy_kwh = current_fuel_consumption * CONVERSION_FACTORS[current_fuel]

    # Retrieve emission factors
    current_emission_factor = EMISSION_FACTORS.get(current_fuel, 0)

    # Calculate current emissions
    current_emissions_kg = current_energy_kwh * current_emission_factor

    # Calculate future emissions
    if mixed_future_sources:
        future_emission_factor = sum(EMISSION_FACTORS[source] * fraction for source, fraction in mixed_future_sources.items())
    else:
        future_emission_factor = EMISSION_FACTORS.get(future_system, 0)

    future_emissions_kg = future_energy_consumption * future_emission_factor

    # Calculate CO₂ savings
    annual_reduction_kg = current_emissions_kg - future_emissions_kg
    annual_carbon_credits_tons = annual_reduction_kg / 1000  # Convert kg to metric tons
    lifetime_carbon_credits_tons = annual_carbon_credits_tons * CREDITING_PERIOD_YEARS

    # Financial calculations
    annual_value_eur = annual_carbon_credits_tons * CARBON_PRICE_PER_TON
    lifetime_value_eur = lifetime_carbon_credits_tons * CARBON_PRICE_PER_TON

    # Format output
    report = f"""
    CARBON_CREDIT_CALCULATION_REPORT
    -----------------------------------
    DOCUMENT_ID: {document_id}
    DATE_ASSESSED: {date_assessed}

    BUILDING_DATA
    -----------------------------------
    BUILDING_TYPE: {building_type}
    LOCATION: {location}
    AREA_SQM: {area_sqm}
    CONSTRUCTION_YEAR: {construction_year}
    HEATING_SYSTEM: {heating_system}

    EMISSIONS_DATA
    -----------------------------------
    CURRENT_FUEL_TYPE: {current_fuel}
    CURRENT_FUEL_CONSUMPTION: {current_fuel_consumption} ({current_fuel} units)
    CURRENT_ENERGY_CONSUMPTION_KWH: {current_energy_kwh} kWh
    CURRENT_EMISSION_FACTOR: {current_emission_factor} kg CO₂/kWh
    CURRENT_EMISSIONS_TOTAL_KG: {current_emissions_kg} kg CO₂

    FUTURE_SYSTEM: {future_system}
    FUTURE_ENERGY_CONSUMPTION: {future_energy_consumption} kWh
    FUTURE_EMISSION_FACTOR: {future_emission_factor} kg CO₂/kWh
    FUTURE_EMISSIONS_TOTAL_KG: {future_emissions_kg} kg CO₂

    CARBON CREDITS
    -----------------------------------
    ANNUAL_REDUCTION_KG: {annual_reduction_kg} kg CO₂
    ANNUAL_CARBON_CREDITS_TONS: {annual_carbon_credits_tons} t CO₂
    CREDITING_PERIOD_YEARS: {CREDITING_PERIOD_YEARS}
    LIFETIME_CARBON_CREDITS_TONS: {lifetime_carbon_credits_tons} t CO₂

    FINANCIAL VALUE
    -----------------------------------
    CARBON_PRICE_EUR_PER_TON: {CARBON_PRICE_PER_TON}
    ANNUAL_VALUE_EUR: {annual_value_eur} €
    LIFETIME_VALUE_EUR: {lifetime_value_eur} €

    RECOMMENDATIONS
    -----------------------------------
    Consider using certified green electricity or adding a photovoltaic system to further reduce emissions and increase carbon credits.
    """

    return report

# Example usage: Gas → Heat Pump Conversion
example_report = calculate_carbon_credits(
    document_id="Sanierungsfahrplan_Bock",
    building_type="Residential",
    location="Germany",
    area_sqm=142,
    construction_year=1997,
    heating_system="Gas (Baujahr 1997 – Gas-NT)",
    current_fuel="natural gas",
    current_fuel_consumption=11000,  # m³
    future_system="heat pump (electricity mix)",
    future_energy_consumption=5500  # kWh
)

print(example_report)
